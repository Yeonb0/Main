- [[문자열]] 출력 시 \`(백틱) 을 이용해 중간에 변수 or 함수 포함해 입력
```js
// 두 문자열을 줄바꿈을 포함해서 출력
console.log(`string text line 1
string text line 2`);

let a = 5;
let b = 10;
console.log(`Fifteen is ${a + b} and not ${2 * a + b}.`);
// "Fifteen is 15 and not 20."

let name = 'Yena Lee';
const greeting = `Hello my name is ${name}`;
console.log(greeting);
// "Hello my name is Yena Lee"

// 중첩 템플릿
const classes = `header ${
  isLargeScreen() ? "icon-expander" : "icon-collapser"
}`;
// 여기서 isLargeScreen()은 true나 false를 반환하는 함수라고 가정합니다!
// isLargeScreen()이 true면 icon-expander false면 icon-collapser 반환
```
